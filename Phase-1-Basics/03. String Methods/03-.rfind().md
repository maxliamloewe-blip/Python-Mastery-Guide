# rfind()


# Function: 

#The rfind() operator returns the last occurence of a value 

# Syntax & Execution;

#First index is 0 ( 1 )

#Only works on strings ( 2 )

#Diffrence with capital letters ( 3 )

#Returns -1 if nothing is found ( 4 )

# Example 

print("Holo".rfind("o"))

#Prints 3

word = "Better"

print(word.rfind("e"))

#Prints 4
