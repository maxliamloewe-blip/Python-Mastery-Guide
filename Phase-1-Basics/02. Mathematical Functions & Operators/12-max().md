# max()


# Function:

#The max() Operator, shows the highest numbers.

# Syntax & Execution:

#Negative numbers are smaller than 0 ( 1 )

#Works for strings aswell ( 2 )

# Example:

print(max(1,2,3))

#Prints 3

#( 1 )

y,x = -100,1

print(max(y,x))

#Prints 1

#( 2 )

print(max("Hello","Hi"))
